STABLE=('st_dev','st_ino','st_mode','st_uid','st_gid','st_nlink','st_size','st_mtime_ns','st_ctime_ns')
def read(path):
    path=Path(path)
    assert path.is_absolute() and path.resolve(strict=True)==path
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC)
    try:
        before=os.fstat(fd)
        assert stat.S_ISREG(before.st_mode) and before.st_uid==before.st_gid==1000 and before.st_nlink==1 and stat.S_IMODE(before.st_mode)==0o600
        with os.fdopen(fd,'rb',closefd=False) as stream:raw=stream.read()
        assert len(raw)==before.st_size and all(getattr(before,key)==getattr(os.fstat(fd),key)==getattr(path.lstat(),key) for key in STABLE)
        return raw
    finally:os.close(fd)
def bound(record,path=None):
    assert type(record) is dict and set(record)=={'path','sha256','size'}
    if path is not None:assert record['path']==str(path)
    raw=read(record['path']);assert hashlib.sha256(raw).hexdigest()==record['sha256'] and len(raw)==record['size']
    return raw
