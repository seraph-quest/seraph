def require(condition, code):
    if not condition:
        raise RuntimeError(code)

def read(path, *, mode=None):
    path = Path(path)
    require(path.is_absolute() and path.resolve(strict=True) == path, 'canonical_read_required')
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        before = os.fstat(fd)
        require(stat.S_ISREG(before.st_mode) and before.st_uid == 1000
            and before.st_gid == 1000 and before.st_nlink == 1, 'private_owned_regular_required')
        require(mode is None or stat.S_IMODE(before.st_mode) == mode, 'original_mode_required')
        chunks = []
        while True:
            chunk = os.read(fd, 1048576)
            if not chunk:
                break
            chunks.append(chunk)
        raw = b''.join(chunks)
        after = os.fstat(fd)
        current = path.lstat()
        require(len(raw) == before.st_size and all(getattr(before, name) == getattr(after, name)
            == getattr(current, name) for name in STABLE), 'read_changed')
        return raw, before
    finally:
        os.close(fd)

def load(path, *, mode=0o600):

    return {'path': str(path), 'sha256': hashlib.sha256(raw).hexdigest(), 'size': len(raw)}

def arbitrary_read(path):
    # Original system/cache aliases are resolved and validated separately.
    path = Path(path).resolve(strict=True)
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        before = os.fstat(fd)
        require(stat.S_ISREG(before.st_mode), 'original_regular_input_required')
        chunks = []
        while True:
            chunk = os.read(fd, 1048576)
            if not chunk:
                break
            chunks.append(chunk)
        raw = b''.join(chunks)
        after = os.fstat(fd)
        current = path.lstat()
        require(len(raw) == before.st_size and all(getattr(before, n) == getattr(after, n)
            == getattr(current, n) for n in STABLE), 'original_input_changed')
        return raw, before
    finally:
        os.close(fd)

def hash_input(path, digest):
    raw, info = arbitrary_read(path)
    require(hashlib.sha256(raw).hexdigest() == digest, 'original_input_hash_drift')
    return info
