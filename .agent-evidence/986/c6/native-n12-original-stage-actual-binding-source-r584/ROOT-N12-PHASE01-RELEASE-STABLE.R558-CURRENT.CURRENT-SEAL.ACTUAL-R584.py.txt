from pathlib import Path
import hashlib,json,os,stat
root=Path('/home/pawel/repos/seraph/.agent-evidence/986/c6')
packet=root/'native-managed-running-case-successor-source-r575'
assert os.getuid()==1000
STABLE=('st_dev','st_ino','st_mode','st_uid','st_gid','st_nlink','st_size','st_mtime_ns','st_ctime_ns')
def identity(info): return tuple(getattr(info,name) for name in STABLE)

def read(path):
    before=path.lstat()
    assert stat.S_ISREG(before.st_mode) and before.st_uid==1000 and before.st_gid==1000 and before.st_nlink==1 and stat.S_IMODE(before.st_mode)==0o600
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC)
    try:
        info=os.fstat(fd)
        assert identity(info)==identity(before)
        with os.fdopen(fd,'rb',closefd=False) as stream: raw=stream.read()
        assert identity(os.fstat(fd))==identity(info)==identity(path.lstat()) and len(raw)==info.st_size
        return raw
    finally: os.close(fd)

def binding(path):
    raw=read(path)
    return {'path':str(path),'sha256':hashlib.sha256(raw).hexdigest(),'size':len(raw)}
